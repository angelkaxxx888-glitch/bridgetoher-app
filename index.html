function loadChecklistState() { const saved = state.checks || []; document.querySelectorAll('.check-item').forEach((item, idx) => { const label = item.querySelector('.check-label')?.textContent.trim(); if (saved.includes(label)) { item.classList.add('checked'); } else { item.classList.remove('checked'); } }); } function saveChecklistState() { const checks = []; document.querySelectorAll('.check-item.checked').forEach(item => { const label = item.querySelector('.check-label')?.textContent.trim(); if (label) checks.push(label); }); state.checks = checks; saveState(); }
